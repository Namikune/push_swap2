static void	push_b_to_a_target(t_context *context)
{
	t_node	*target;

	target = context->b.head->target_node;
	while (context->a.head != target)
	{
		if (get_node_position(&context->a, target) <= context->a.size / 2)
			ra(context);
		else
			rra(context);
	}
	pa(context);
}

static void	final_alignment(t_context *context)
{
	t_node	*min_node;

	min_node = find_min(&context->a);
	while (context->a.head != min_node)
	{
		if (get_node_position(&context->a, min_node) <= context->a.size / 2)
			ra(context);
		else
			rra(context);
	}
}

void	complex_sort(t_context *context)
{
	rank_numbers(&context->a);
	if (context->a.size > 3 && !stack_is_sorted(&context->a))
		pb(context);
	if (context->a.size > 3 && !stack_is_sorted(&context->a))
		pb(context);
	while (context->a.size > 3 && !stack_is_sorted(&context->a))
	{
		set_target_a_to_b(&context->a, &context->b);
		calculate_costs(&context->a, &context->b);
		set_cheapest(&context->a);
		push_cheapest_to_b(context);
	}
	if (!stack_is_sorted(&context->a))
		sort_three(context);
	while (context->b.size > 0)
	{
		set_target_b_to_a(&context->b, &context->a);
		push_b_to_a_target(context);
	}
	final_alignment(context);
}

int	error_syntax(char *str)
{
	int	i;

	i = 0;
	if (!str || !*str)
		return (1);
	if (str[i] == '+' || str[i] == '-')
	{
		if (!ft_isdigit(str[i + 1]))
			return (1);
		i++;
	}
	while (str[i])
	{
		if (!ft_isdigit(str[i]))
			return (1);
		i++;
	}
	return (0);
}

int	error_duplicate(t_stack *a, int n)
{
	t_node	*current;

	if (!a || !a->head)
		return (0);
	current = a->head;
	while (current)
	{
		if (current->value == n)
			return (1);
		current = current->next;
	}
	return (0);
}

void	free_errors(t_context *context)
{
	stack_free(&context->a);
	stack_free(&context->b);
	write(2, "Error\n", 6);
	exit(1);
}

{
	if (size < 100)
		return (15);
	else if (size <= 250)
		return (25);
	else
		return (35);
}

static void	push_chunks_to_b(t_context *context, int chunk_size)
{
	int	i;

	i = 0;
	while (context->a.size > 0)
	{
		if (context->a.head->index <= i)
		{
			pb(context);
			rb(context);
			i++;
		}
		else if (context->a.head->index <= i + chunk_size)
		{
			pb(context);
			i++;
		}
		else
			ra(context);
	}
}

static void	push_back_to_a(t_context *context)
{
	t_node	*max_node;
	int		pos;

	while (context->b.head != NULL)
	{
		max_node = find_max(&context->b);
		pos = get_node_position(&context->b, max_node);
		while (context->b.head != max_node)
		{
			if (pos <= (context->b.size / 2))
				rb(context);
			else
				rrb(context);
		}
		pa(context);
	}
}

void	medium_sort(t_context *context)
{
	int	chunk_size;

	rank_numbers(&context->a);
	chunk_size = get_chunk_size(context->a.size);
	push_chunks_to_b(context, chunk_size);
	push_back_to_a(context);
}

long	ft_atol(const char *s, int *error)
{
	long	result;
	int		sign;

	result = 0;
	sign = 1;
	while (*s == ' ' || (*s >= 9 && *s <= 13))
		s++;
	if (*s == '+' || *s == '-')
	{
		if (*s == '-')
			sign = -1;
		s++;
	}
	if (!ft_isdigit(*s))
		return (*error = 1, 0);
	while (ft_isdigit(*s))
	{
		result = result * 10 + (*s - '0');
		if ((result * sign) > INT_MAX || (result * sign) < INT_MIN)
			return (*error = 1, 0);
		s++;
	}
	if (*s != '\0')
		return (*error = 1, 0);
	return (result * sign);
}

void	init_stack_a(t_context *context, char **argv, int start_idx)
{
	long	n;
	int		i;
	int		error;
	t_node	*new_node;

	i = 0;
	while (argv[i])
		i++;
	while (--i >= start_idx)
	{
		if (ft_strncmp(argv[i], "--", 2) == 0)
			continue ;
		error = 0;
		if (error_syntax(argv[i]))
			free_errors(context);
		n = ft_atol(argv[i], &error);
		if (error == 1 || error_duplicate(&context->a, (int)n))
			free_errors(context);
		new_node = node_new((int)n);
		if (!new_node)
			free_errors(context);
		stack_push_top(&context->a, new_node);
	}
}

static int	ft_abs(int n)
{
	if (n < 0)
		return (-n);
	return (n);
}

static int	get_max(int a, int b)
{
	if (a > b)
		return (a);
	return (b);
}

void	calculate_costs(t_stack *a, t_stack *b)
{
	t_node	*curr;
	int		pos_a;
	int		pos_b;

	curr = a->head;
	while (curr)
	{
		pos_a = get_node_position(a, curr);
		if (pos_a > a->size / 2)
			pos_a -= a->size;
		pos_b = get_node_position(b, curr->target_node);
		if (pos_b > b->size / 2)
			pos_b -= b->size;
		if ((pos_a >= 0 && pos_b >= 0) || (pos_a < 0 && pos_b < 0))
			curr->push_cost = get_max(ft_abs(pos_a), ft_abs(pos_b));
		else
			curr->push_cost = ft_abs(pos_a) + ft_abs(pos_b);
		curr = curr->next;
	}
}

void	set_cheapest(t_stack *a)
{
	long	min_cost;
	t_node	*curr;
	t_node	*cheapest_node;

	min_cost = LONG_MAX;
	cheapest_node = NULL;
	curr = a->head;
	while (curr)
	{
		curr->cheapest = false;
		if (curr->push_cost <= min_cost)
		{
			min_cost = curr->push_cost;
			cheapest_node = curr;
		}
		curr = curr->next;
	}
	if (cheapest_node)
		cheapest_node->cheapest = true;
}

{
	t_node	*cheap;
	int		pos_a;
	int		pos_b;

	cheap = context->a.head;
	while (cheap && !cheap->cheapest)
		cheap = cheap->next;
	pos_a = get_node_position(&context->a, cheap);
	pos_b = get_node_position(&context->b, cheap->target_node);
	if (pos_a <= context->a.size / 2 && pos_b <= context->b.size / 2)
	{
		while (context->a.head != cheap
			&& context->b.head != cheap->target_node)
			rr(context);
	}
	else if (pos_a > context->a.size / 2 && pos_b > context->b.size / 2)
	{
		while (context->a.head != cheap
			&& context->b.head != cheap->target_node)
			rrr(context);
	}
	rotate_a_to_top(context, cheap);
	rotate_b_to_top(context, cheap->target_node);
	pb(context);
}

{
	t_node	*curr_b;
	t_node	*target;
	int		best_match;

	best_match = -1;
	target = NULL;
	curr_b = b->head;
	while (curr_b)
	{
		if (curr_b->index < node_a->index && curr_b->index > best_match)
		{
			best_match = curr_b->index;
			target = curr_b;
		}
		curr_b = curr_b->next;
	}
	if (best_match == -1)
		return (find_max(b));
	return (target);
}

		curr_a = curr_a->next;

/* A static helper that counts the "mistakes" */
static int	mistakes_count(t_context *context)
{
	t_node	*start;
	t_node	*next;
	int		i;

	start = context->a.head;
	i = 0;
	while (start->next != NULL)
	{
		next = start->next;
		while (next != NULL)
		{
			if (start->value > next->value)
				i++;
			next = next->next;
		}
		start = start->next;
	}
	return (i);
}

/*
Calculates disorder via its formula:
Disorder  =  mistakes × 10,000 / total pairs
Times 10,000 to turn it into an int for bench to print properly
*/
long	disorder(t_context	*context)
{
	long	size;
	long	result;
	long	mistakes;

	if (context->a.size < 2)
		return (0);
	size = context->a.size;
	mistakes = mistakes_count(context);
	result = ((mistakes * 10000) / (((size * (size - 1)) / 2)));
	return (result);
}

/*
Creates a new node
index = -1 means "no rank yet"
The Turk fields start at zero, false or NULL
*/
t_node	*node_new(int value)
{
	t_node	*node;

	node = malloc(sizeof(t_node));
	if (!node)
		return (NULL);
	node->value = value;
	node->index = -1;
	node->push_cost = 0;
	node->above_median = false;
	node->cheapest = false;
	node->target_node = NULL;
	node->next = NULL;
	node->prev = NULL;
	return (node);
}

/*
Puts the node on top of the stack and adds 1 to size
The old top (if there is one) now has the new node above it
*/
void	stack_push_top(t_stack *stack, t_node *node)
{
	node->prev = NULL;
	node->next = stack->head;
	if (stack->head != NULL)
		stack->head->prev = node;
	stack->head = node;
	stack->size += 1;
}

/*
-Detaches and returns the stack's top node
-The new top has no node above it anymore
-Subtracts 1 from size
-Returns NULL if stack is empty
*/
t_node	*stack_pop_top(t_stack *stack)
{
	t_node	*top_node;

	if (stack->head == NULL)
		return (NULL);
	top_node = stack->head;
	stack->head = top_node->next;
	if (stack->head != NULL)
		stack->head->prev = NULL;
	top_node->next = NULL;
	top_node->prev = NULL;
	stack->size -= 1;
	return (top_node);
}

/*
-Returns the stack's bottom node
-Returns NULL if stack is empty
*/
t_node	*stack_last(t_stack *stack)
{
	t_node	*last_node;

	if (!stack)
		return (NULL);
	last_node = stack->head;
	if (last_node == NULL)
		return (NULL);
	while (last_node->next != NULL)
	{
		last_node = last_node->next;
	}
	return (last_node);
}

/*
Frees every node in the stack
Sets stack's size to zero
*/
void	stack_free(t_stack *stack)
{
	t_node	*node;

	node = stack->head;
	while (stack->head != NULL)
	{
		node = node->next;
		free(stack->head);
		stack->head = node;
	}
	stack->size = 0;
}

{
	t_node	*node;
	t_node	*next_node;

	if (!stack->head)
		return (1);
	node = stack->head;
	next_node = node->next;
	while (next_node != NULL)
	{
		if (next_node->value >= node->value)
		{
			node = next_node;
			next_node = next_node->next;
		}
		else
			return (0);
	}
	return (1);
}

/*

TWEAK THE DISORDER BORDERS DEPENDING ON PERFORMANCE!!!

static void	disorder_picked_strategy(t_context *context)
{
	long		disorder_value;

	disorder_value = context->stats.disorder_value;
	if (disorder_value < 2000)
		context->config.chosen_strategy = SIMPLE;
	else if (disorder_value < 5000)
		context->config.chosen_strategy = MEDIUM;
	else
		context->config.chosen_strategy = COMPLEX;
}

/*Chooses the sort that is saved in config*/
static void	choose_sort(t_context *context)
{
	if (context->config.chosen_strategy == SIMPLE)
		simple_sort(context);
	else if (context->config.chosen_strategy == MEDIUM)
		medium_sort(context);
	else if (context->config.chosen_strategy == COMPLEX)
		complex_sort(context);
}

/*
Runs the strategy:
-if sorted, returns nothing (0 or 1 number)
-if 2 numbers, swap a (sa)
-if 3 numbers: sort_three()
-if 4 or more, decides with context's chosen strategy
*/
void	dispatch_sort(t_context *context)
{
	t_stack		*a;
	t_strategy	strategy;

	a = &context->a;
	strategy = context->config.strategy;
	context->stats.disorder_value = disorder(context);
	context->config.chosen_strategy = SIMPLE;
	if (stack_is_sorted(a) == 1)
		return ;
	if (a->size == 2)
	{
		sa(context);
		return ;
	}
	if (a->size == 3)
	{
		sort_three(context);
		return ;
	}
	if (strategy == ADAPTIVE)
		disorder_picked_strategy(context);
	else
		context->config.chosen_strategy = strategy;
	choose_sort(context);

#include "push_swap.h"

static void	init_context(t_context *context)
{
	int	i;

	context->a.head = NULL;
	context->a.size = 0;
	context->b.head = NULL;
	context->b.size = 0;
	context->config.strategy = ADAPTIVE;
	context->config.bench = 0;
	context->config.chosen_strategy = ADAPTIVE;
	context->stats.total = 0;
	context->stats.disorder_value = 0;
	i = 0;
	while (i < 11)
		context->stats.ops[i++] = 0;
}

void	free_split(char **split_arr)
{
	int	i;

	i = 0;
	if (!split_arr)
		return ;
	while (split_arr[i])
	{
		free(split_arr[i]);
		i++;
	}
	free(split_arr);
}

static char	**get_args(int argc, char **argv, int *allocated, int *start_idx)
{
	char	**args;

	*allocated = 0;
	if (argc == 2)
	{
		args = ft_split(argv[1], ' ');
		*allocated = 1;
		*start_idx = 0;
	}
	else
	{
		args = argv;
		*start_idx = 1;
	}
	return (args);
}

int	main(int argc, char **argv)
{
	t_context	context;
	char		**args;
	int			is_allocated;
	int			start_idx;

	if (argc == 1 || (argc == 2 && !argv[1][0]))
		return (1);
	init_context(&context);
	parse_flags(argc, argv, &context);
	args = get_args(argc, argv, &is_allocated, &start_idx);
	init_stack_a(&context, args, start_idx);
	if (is_allocated)
		free_split(args);
	dispatch_sort(&context);
	if (context.config.bench == 1)
		bench(&context);
	stack_free(&context.a);
	return (0);
}
